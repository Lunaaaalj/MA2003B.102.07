\begin{frame}{Objetivo y tratamiento de los datos}
  \scriptsize

  \textbf{Objetivo principal.} Clasificar cada \textbf{día-estación} como excedencia / no excedencia del MDA8 de ozono al umbral de \textbf{51 ppb} (meta a 5 años de la NOM-020-SSA1-2021), a partir de los precursores y la meteorología del \textbf{mismo día}. Modelo \textbf{explicativo, no de pronóstico}. Criterio de logro: AUC $\geq 0.75$ en validación, con sensibilidad y especificidad por separado; la exactitud se descarta por el desbalance de clases.
  \vspace{2pt}

  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \textbf{Tratamiento de los datos}
      \begin{itemize}[itemsep=1pt,topsep=1pt,parsep=0pt,leftmargin=*]
        \item \textbf{Imputación}: 1.67\,\% de las celdas (144{,}160 de
          8{,}615{,}145) por interpolación temporal lineal, límite de 3 h, sin
          extrapolar y por estación. Los huecos largos no se imputan.
        \item \textbf{Atípicos: se detectan y se conservan.} El objetivo es
          clasificar excedencias; recortar los extremos eliminaría justo la
          información relevante. Quedan marcados con indicadoras.
        \item \textbf{Sí se eliminan los errores de medición}: centinelas
          $-9999$ del datalogger, topes de saturación 999/1000/1001 en PM y
          valores fuera del rango físico del sensor (1{,}359 celdas en total,
          0.0158\,\%).
      \end{itemize}
      \vspace{1pt}
      \textbf{ACP.} Ante la multicolinealidad entre precursores
      (\texttt{NO}--\texttt{NOX}: $r = 0.94$) se retienen 3 componentes ---carga
      primaria de combustión, forzamiento fotoquímico y estancamiento--- con
      \textbf{55.9\,\%} de la varianza de los 15 predictores; sus \emph{scores}
      entran a los modelos.
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \includegraphics[width=\linewidth]{pca_scree.pdf}

      \scriptsize Varianza explicada por componente.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Regresión logística y discriminante lineal}
  \scriptsize

  \textbf{Partición.} Entrenamiento 2021--2024, validación 2025.
  Predictores: \texttt{PC1}--\texttt{PC3} más 12 indicadoras
  (\texttt{zona}, \texttt{tipo\_dia}, \texttt{temporada}), con niveles de
  referencia \texttt{Noreste}, día laborable e invierno.
  \vspace{1pt}

  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \textbf{Regresión logística} (umbral de Youden $=0.288$)
      \begin{itemize}[itemsep=1pt,topsep=2pt,parsep=0pt,leftmargin=*]
        \item Razones de momios por DE: \texttt{PC1} $=1.47$,
          \texttt{PC2} $=1.58$, \texttt{PC3} $=1.22$
        \item Indicadoras: \texttt{temporada\_primavera} $=5.11$,
          \texttt{temporada\_otono} $=4.25$, \texttt{zona\_Noroeste}
          $=3.35$, \texttt{zona\_Suroeste} $=3.18$
        \item AUC $=\mathbf{0.893}$; sensibilidad $=0.831$;
          especificidad $=0.824$
      \end{itemize}
      \vspace{1pt}
      \textbf{Discriminante lineal} (Youden $=0.293$)
      \begin{itemize}[itemsep=1pt,topsep=2pt,parsep=0pt,leftmargin=*]
        \item Normalidad: sesgos Box--Cox $<0.5$; se sostiene
        \item Homocedasticidad: \textbf{no se cumple} (razones de DE
          0.88/0.69/0.92; varianza generalizada $19.6 \to 6.1$)
        \item AUC $=\mathbf{0.893}$; sensibilidad $=0.845$;
          especificidad $=0.811$
      \end{itemize}
      \vspace{1pt}
      \textbf{Comparación.} El AUC coincide a cuatro decimales (0.8928 en
      ambos), la prueba de DeLong no detecta diferencia
      ($Z=0.011$, $p=0.99$) y las clasificaciones coinciden en el
      \textbf{98.1\,\%} de los días-estación. El orden y el signo de los
      coeficientes del discriminante reproducen los de la logística.
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=\linewidth]{roc_modelos.pdf}

      \scriptsize Las dos curvas ROC se superponen casi por completo.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusiones hasta el momento}
  \footnotesize

  \begin{enumerate}[label=\textbf{\roman*.},itemsep=3pt,topsep=0pt,parsep=0pt,leftmargin=*]
    \item \textbf{Los dos modelos son equivalentes.} Sus AUC son
      aproximadamente iguales: la frontera de decisión está bien delimitada
      por los datos y no por el método, y la heterocedasticidad documentada
      en el discriminante es real pero no altera el desempeño.

    \item \textbf{La estacionalidad domina sobre las demás condiciones.}
      \texttt{temporada\_primavera} alcanza momios de \textbf{5.11},
      por encima de cualquier componente o zona. La excedencia de ozono es
      un fenómeno principalmente estacional.

    \item \textbf{Hay una señal espacial que no está medida.} El Noroeste y
      el Suroeste presentan alrededor de \textbf{tres veces} los momios del
      Noreste aun con las mismas condiciones meteorológicas: una
      característica de la zona misma ---no capturada por las variables
      disponibles--- las convierte en zonas de riesgo.

    \item \textbf{El calendario laboral casi no tiene efecto.} Solo
      \texttt{tipo\_dia\_domingo} resulta significativo, y de forma
      relativamente baja ($1.39$); \texttt{sabado} y \texttt{festivo} no lo
      son. Un experimento de contraste del efecto fin de semana sería, por
      lo tanto, un análisis débil.
  \end{enumerate}
\end{frame}
